export default async function handler(req, res) { if (req.method !== 'POST') return res.status(405).json({ error: 'Method not allowed' }); const { day, intensity, mood } = req.body; if (!day || day.trim().length < 10) return res.status(400).json({ error: 'Tell me more about your day!' }); try { const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': process.env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', }, body: JSON.stringify({ model: 'claude-sonnet-4-20250514', max_tokens: 500, messages: [{ role: 'user', content: `You are a comedian doing a roast. Roast this person's day. Intensity level: ${intensity || 'medium'}. Their mood today: ${mood || '🙂'}. Be specific, funny, creative. Under 130 words. End with ONE surprisingly kind sentence. No emojis. No hashtags.\n\nTheir day: ${day}` }] }) }); const data = await response.json(); if (!response.ok) throw new Error(data.error?.message || 'API error'); return res.status(200).json({ roast: data.content[0].text }); } catch (error) { console.error(error); return res.status(500).json({ error: 'AI took one look at your day and gave up too. Classic.' }); } }
